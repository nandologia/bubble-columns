[Bubble Columns]

#    Highest water column a single soul sand or magma block can drive, in nodes.
bubble_columns_max_height (Maximum column height) int 24 1 128

#    Terminal upward speed inside a soul sand column, in nodes per second.
bubble_columns_up_speed (Updraft speed) float 8.0 0.0 40.0

#    Terminal downward speed inside a magma whirlpool, in nodes per second.
bubble_columns_down_speed (Whirlpool speed) float 6.0 0.0 40.0

#    How sharply mobs, boats and dropped items are pulled to that speed, in
#    nodes per second squared. Does not affect players, who are moved by the
#    gravity multipliers below instead.
bubble_columns_accel (Entity acceleration) float 30.0 1.0 200.0

#    Gravity multiplier for a player in an updraft. 0 neutralises gravity so
#    it cannot claw back what the lift gains between steps. Do NOT set this
#    negative expecting it to lift: gravity overrides have almost no effect on
#    a player inside a liquid, where the client uses its own liquid movement
#    model. The lifting is done by the speed setting above.
bubble_columns_up_gravity (Updraft gravity multiplier) float 0.0 -2.0 1.0

#    Refill the player's air while they ride a soul sand column, as in
#    Minecraft. Magma whirlpools never refill air whatever this is set to.
bubble_columns_restore_air (Updraft refills air) bool true

#    Let Soul Soil make updrafts as well as Soul Sand. Minecraft does not do
#    this -- only soul sand works there -- but the two blocks look alike and
#    are easy to mix up, so enable this if you would rather both worked.
bubble_columns_soul_soil_too (Soul soil makes columns too) bool false

#    Trace column detection and physics to debug.txt. Very noisy; for
#    diagnosing a column that does not seem to do anything.
bubble_columns_debug (Debug logging) bool false
