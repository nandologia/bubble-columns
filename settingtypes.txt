[Bubble Columns]

#    How fast an updraft carries you up. More negative is faster; it is a
#    multiplier on how fast you would otherwise sink.
#    Easier to judge by riding one than by guessing: /bubblespeed changes it
#    without a restart.
bubble_columns_liquid_sink (Updraft climb speed) float -1.4 -6.0 0.0

#    How fast a whirlpool drags you down. Higher is faster.
bubble_columns_down_sink (Whirlpool sink speed) float 2.0 1.0 6.0

#    How long a boat caught in a whirlpool rocks before it goes under, in
#    seconds. The rocking is the warning: paddle clear in time and the boat is
#    fine. Lower is crueller.
bubble_columns_boat_rock_time (Boat rocking time) float 3.0 0.0 30.0

#    Refill your air while you are in a column, so you cannot drown in one.
#    Turning this off makes whirlpools far more dangerous.
bubble_columns_restore_air (Columns refill your air) bool true

#    Froth on the water above a column, so you can spot one from a boat or the
#    shore. Purely decorative.
bubble_columns_surface_bubbles (Bubbling at the surface) bool true

#    Let Soul Soil make updrafts as well as Soul Sand. Off by default: only
#    soul sand should work. The two blocks look alike and sit near each other
#    in the inventory, so enable this if you would rather both worked.
bubble_columns_soul_soil_too (Soul soil makes columns too) bool false

#    Log column detection and physics to debug.txt. Very noisy; for working
#    out why a column does not seem to do anything.
bubble_columns_debug (Debug logging) bool false
